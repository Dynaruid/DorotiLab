cmake_minimum_required(VERSION 3.24)
project(doroti_qt_quick_validation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.6 REQUIRED COMPONENTS Core Gui Widgets Quick Qml Test WebEngineQuick)
find_package(Vulkan REQUIRED)
set(DOROTI_SHIM "" CACHE FILEPATH "Path to the built product libdoroti_qt_host.so")
if(NOT EXISTS "${DOROTI_SHIM}")
  message(FATAL_ERROR "Set DOROTI_SHIM to the product native library")
endif()
set(native "${CMAKE_CURRENT_SOURCE_DIR}/../../../DorotiTestbedApp/linux/native")
add_library(product-driver SHARED product-driver.cpp gpu-driver.cpp)
add_executable(native-contract native-contract.cpp)
foreach(target product-driver native-contract)
  target_compile_definitions(${target} PRIVATE DOROTI_QT_HOST_BUILD DOROTI_QT_QUICK)
  target_include_directories(${target} PRIVATE "${native}/include")
  target_link_libraries(${target} PRIVATE "${DOROTI_SHIM}" Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Quick Qt6::Qml Qt6::Test Qt6::WebEngineQuick Vulkan::Headers)
endforeach()
