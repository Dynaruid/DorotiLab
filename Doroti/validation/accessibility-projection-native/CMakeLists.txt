cmake_minimum_required(VERSION 3.24)
project(accessibility_projection LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
if(WIN32)
  add_executable(accessibility_projection windows.cpp ../../src/Doroti.Host.WindowsAppSdk.Native/src/accessibility_bridge.cpp)
  target_compile_definitions(accessibility_projection PRIVATE NOMINMAX UNICODE _UNICODE)
  target_link_libraries(accessibility_projection PRIVATE uiautomationcore ole32 oleaut32 user32)
else()
  find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Widgets OpenGL OpenGLWidgets)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(WaylandClient REQUIRED IMPORTED_TARGET wayland-client)
  find_program(WAYLAND_SCANNER wayland-scanner REQUIRED)
  set(native "${CMAKE_CURRENT_SOURCE_DIR}/../../../DorotiTestbedApp/linux/native")
  foreach(protocol IN ITEMS ext-background-effect-v1 kde-blur)
    execute_process(COMMAND "${WAYLAND_SCANNER}" client-header "${native}/protocols/${protocol}.xml" "${CMAKE_CURRENT_BINARY_DIR}/${protocol}-client-protocol.h" COMMAND_ERROR_IS_FATAL ANY)
    execute_process(COMMAND "${WAYLAND_SCANNER}" private-code "${native}/protocols/${protocol}.xml" "${CMAKE_CURRENT_BINARY_DIR}/${protocol}-protocol.c" COMMAND_ERROR_IS_FATAL ANY)
    list(APPEND protocol_sources "${CMAKE_CURRENT_BINARY_DIR}/${protocol}-protocol.c")
  endforeach()
  add_executable(accessibility_projection qt.cpp ${protocol_sources})
  target_include_directories(accessibility_projection PRIVATE "${native}/include" "${CMAKE_CURRENT_BINARY_DIR}")
  target_link_libraries(accessibility_projection PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::OpenGL Qt6::OpenGLWidgets PkgConfig::WaylandClient)
endif()
enable_testing()
add_test(NAME accessibility_projection COMMAND accessibility_projection)
set_tests_properties(accessibility_projection PROPERTIES TIMEOUT 1200)
if(NOT WIN32)
  set_tests_properties(accessibility_projection PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
