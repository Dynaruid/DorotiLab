cmake_minimum_required(VERSION 3.24)
project(DorotiQtInterleavingSpike LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Test)
find_package(PkgConfig REQUIRED)
pkg_check_modules(XCapture REQUIRED IMPORTED_TARGET x11 xcomposite)
add_executable(widget-shell probe.cpp)
target_link_libraries(widget-shell PRIVATE Qt6::Widgets Qt6::Test PkgConfig::XCapture)
option(DOROTI_PROBE_WEBENGINE "Build the optional live WebEngine comparison" OFF)
if(DOROTI_PROBE_WEBENGINE)
  find_package(Qt6 6.8 REQUIRED COMPONENTS WebEngineWidgets)
  add_executable(webengine-shell probe.cpp)
  target_compile_definitions(webengine-shell PRIVATE DOROTI_PROBE_WEBENGINE)
  target_link_libraries(webengine-shell PRIVATE Qt6::Widgets Qt6::Test Qt6::WebEngineWidgets PkgConfig::XCapture)
endif()
